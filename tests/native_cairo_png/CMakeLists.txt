cmake_minimum_required(VERSION 3.20)
project(CHOSCairoPNG LANGUAGES C)
add_executable(native_cairo_png main.c)
target_include_directories(native_cairo_png PRIVATE "${GRAPHICS_PREFIX}/include/cairo" "${PNG_PREFIX}/include")
target_link_libraries(native_cairo_png PRIVATE "${GRAPHICS_PREFIX}/lib/libcairo.a"
    "${GRAPHICS_PREFIX}/lib/libpixman-1.a" "${PNG_PREFIX}/lib/libpng.a" "${ZLIB_PREFIX}/lib/libz.a")
if(FONTCONFIG_PREFIX)
    find_package(Fontconfig CONFIG REQUIRED PATHS "${FONTCONFIG_PREFIX}/lib/cmake/Fontconfig"
        NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
    target_link_libraries(native_cairo_png PRIVATE Fontconfig::Fontconfig)
else()
    target_link_libraries(native_cairo_png PRIVATE "${FONT_PREFIX}/lib/libfreetype.a")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    target_link_options(native_cairo_png PRIVATE "-Wl,--strip-debug")
else()
    find_package(Threads REQUIRED)
    target_link_libraries(native_cairo_png PRIVATE Threads::Threads m)
endif()
