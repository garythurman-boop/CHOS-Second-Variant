cmake_minimum_required(VERSION 3.20)
project(CHOSWTFResource LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${WEBKIT_SOURCE}/Source/WTF/wtf/CPUTime.cpp" OR NOT EXISTS "${ENGINE_BUILD}/cmakeconfig.h")
 message(FATAL_ERROR "Use checked upstream source and real JSCOnly configuration")
endif()
set(WTF_PATH "${WEBKIT_SOURCE}/Source/WTF/wtf")
add_executable(wtf_resource main.cpp ../../ports/webkit/chos/CPUTimeCHOS.cpp ../../ports/webkit/chos/MemoryFootprintCHOS.cpp "${WTF_PATH}/CPUTime.cpp" "${WTF_PATH}/CurrentTime.cpp" "${WTF_PATH}/MonotonicTime.cpp" "${WTF_PATH}/WallTime.cpp")
target_include_directories(wtf_resource PRIVATE "${ENGINE_BUILD}" "${WEBKIT_SOURCE}/Source/WTF" "${WEBKIT_SOURCE}/Source/bmalloc")
target_compile_definitions(wtf_resource PRIVATE BUILDING_JSCONLY__ BUILDING_WITH_CMAKE=1 HAVE_CONFIG_H=1 BUILDING_WTF=1 NDEBUG=1)
