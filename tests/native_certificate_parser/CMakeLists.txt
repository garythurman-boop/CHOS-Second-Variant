cmake_minimum_required(VERSION 3.20)
project(CHOSNativeCertificateParser LANGUAGES C)
if(NOT CERTIFICATE_CASES OR NOT CHOS_ROOT)
    message(FATAL_ERROR "Provide generated CERTIFICATE_CASES and CHOS_ROOT")
endif()
include("${CHOS_ROOT}/ports/webcore-network-types/CertificateParser.cmake")
add_executable(native_certificate_parser main.c)
target_include_directories(native_certificate_parser PRIVATE "${CERTIFICATE_CASES}")
target_link_libraries(native_certificate_parser PRIVATE CHOS::CertificateParser)
target_link_options(native_certificate_parser PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    find_package(Threads REQUIRED)
    target_link_libraries(native_certificate_parser PRIVATE Threads::Threads m)
endif()
