cmake_minimum_required(VERSION 3.20)
project(CHOSNativeText LANGUAGES CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CHOS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT DEFINED TEXT_PREFIX OR NOT DEFINED HB_BUILD OR NOT DEFINED TEXT_CASES_HEADER)
    message(FATAL_ERROR "Provide the built font prefix,HarfBuzz directory and generated vectors")
endif()
add_library(chos-text STATIC "${CHOS_ROOT}/userland/text/text.cpp")
target_include_directories(chos-text PUBLIC "${CHOS_ROOT}/userland/text/include" "${TEXT_PREFIX}/include/freetype2" "${CHOS_ROOT}/third_party/harfbuzz/src" "${HB_BUILD}/src")
if(CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    target_compile_definitions(chos-text PRIVATE CHOS_TEXT_WITH_ICU=1)
else()
    target_compile_definitions(chos-text PRIVATE CHOS_TEXT_WITH_ICU=0)
endif()
add_executable(native_text main.cpp)
if(CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    # Diagnostic resources are real unchanged licensed font files. The normal
    # filesystem fixture remains explicit and never falls back to these bytes.
    set(FONT_ASM "${CMAKE_CURRENT_BINARY_DIR}/font-assets.S")
    file(WRITE "${FONT_ASM}" ".section .rodata.font_assets,\"a\"\n")
    set(FONT_NAMES NotoSans-Regular.ttf NotoNaskhArabic-Regular.ttf NotoSansDevanagari-Regular.ttf 3e46c3b84c1370a06594736c7f8acebf810bbb3b.ttf 872d2955d326bd6676a06f66b8238ebbaabc212f.ttf)
    set(INDEX 0)
    foreach(NAME IN LISTS FONT_NAMES)
        set(FONT_PATH "${CHOS_ROOT}/third_party/text-fixtures/fonts/${NAME}")
        file(APPEND "${FONT_ASM}" ".balign 8\n.global chos_font_${INDEX}_start\n.global chos_font_${INDEX}_end\nchos_font_${INDEX}_start:\n.incbin \"${FONT_PATH}\"\nchos_font_${INDEX}_end:\n")
        set_property(SOURCE "${FONT_ASM}" APPEND PROPERTY OBJECT_DEPENDS "${FONT_PATH}")
        math(EXPR INDEX "${INDEX}+1")
    endforeach()
    target_sources(native_text PRIVATE "${FONT_ASM}")
endif()
target_include_directories(native_text PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
configure_file("${TEXT_CASES_HEADER}" text-cases.h COPYONLY)
if(CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    target_link_libraries(native_text PRIVATE chos-text "-Wl,--start-group" "${HB_BUILD}/libharfbuzz-icu.a" "${HB_BUILD}/libharfbuzz.a" "${TEXT_PREFIX}/lib/libfreetype.a" "${CHOS_SDK_ROOT}/lib/libicuuc.a" "${CHOS_SDK_ROOT}/lib/libicudata.a" "-Wl,--end-group")
    target_link_options(native_text PRIVATE "-Wl,-Map,${CMAKE_BINARY_DIR}/native_text.map" "-Wl,--strip-debug")
else()
    find_package(Threads REQUIRED)
    target_compile_definitions(native_text PRIVATE CHOS_TEXT_HOST=1)
    target_link_libraries(native_text PRIVATE chos-text "${HB_BUILD}/libharfbuzz.a" "${TEXT_PREFIX}/lib/libfreetype.a" Threads::Threads m)
endif()

# Optional real original JavaScriptCore execution in the same isolated client.
if(TEXT_ENGINE_KIT)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
        message(FATAL_ERROR "The pinned original engine objects are AArch64 CHOS objects")
    endif()
    include("${TEXT_ENGINE_OBJECTS}")
    target_sources(native_text PRIVATE javascript.cpp ${TEXT_ENGINE_OBJECT_LIST})
    target_include_directories(native_text PRIVATE "${TEXT_ENGINE_HEADERS}")
    target_compile_definitions(native_text PRIVATE TEXT_WITH_JSC=1)
    target_link_libraries(native_text PRIVATE "-Wl,--start-group" "${TEXT_ENGINE_KIT}/engine/lib/libJavaScriptCore.a" "${TEXT_ENGINE_KIT}/engine/lib/libWTF.a" "${TEXT_ENGINE_KIT}/engine/lib/libbmalloc.a" "${CHOS_SDK_ROOT}/lib/libicui18n.a" "${CHOS_SDK_ROOT}/lib/libicuuc.a" "${CHOS_SDK_ROOT}/lib/libicudata.a" "-Wl,--end-group")
endif()

install(TARGETS chos-text ARCHIVE DESTINATION lib)
install(FILES "${CHOS_ROOT}/userland/text/include/chos/text.h" DESTINATION include/chos)
