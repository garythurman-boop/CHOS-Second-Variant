cmake_minimum_required(VERSION 3.20)
project(CHOSWTFCalendar LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${WEBKIT_SOURCE}/Source/WTF/wtf/DateMath.cpp" OR NOT EXISTS "${ENGINE_BUILD}/cmakeconfig.h")
 message(FATAL_ERROR "Use checked upstream source and real JSCOnly configuration")
endif()
set(WTF_PATH "${WEBKIT_SOURCE}/Source/WTF/wtf")
add_executable(wtf_calendar main.cpp "${WTF_PATH}/DateMath.cpp" "${WTF_PATH}/CurrentTime.cpp" "${WTF_PATH}/MonotonicTime.cpp" "${WTF_PATH}/WallTime.cpp" "${WTF_PATH}/WTFProcess.cpp" "${WTF_PATH}/SIMDUTF.cpp")
target_include_directories(wtf_calendar PRIVATE .. "${ENGINE_BUILD}" "${WEBKIT_SOURCE}/Source/WTF" "${WEBKIT_SOURCE}/Source/bmalloc")
target_compile_definitions(wtf_calendar PRIVATE BUILDING_JSCONLY__ BUILDING_WITH_CMAKE=1 HAVE_CONFIG_H=1 BUILDING_WTF=1 NDEBUG=1)
