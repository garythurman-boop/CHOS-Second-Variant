cmake_minimum_required(VERSION 3.20)
project(CHOSNativeXML LANGUAGES C)
if(NOT XML_PREFIX)
  message(FATAL_ERROR "Provide verified XML_PREFIX")
endif()
add_executable(native_xml main.c)
target_include_directories(native_xml PRIVATE "${XML_PREFIX}/include/libxml2")
target_link_libraries(native_xml PRIVATE "${XML_PREFIX}/lib/libxml2.a")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
  find_package(Threads REQUIRED)
  target_link_libraries(native_xml PRIVATE Threads::Threads m)
else()
  target_link_options(native_xml PRIVATE "-Wl,--strip-debug")
endif()
