cmake_minimum_required(VERSION 3.20)
project(CHOSNativeWebP LANGUAGES C)
if(NOT WEBP_PREFIX OR NOT WEBP_CASES)
    message(FATAL_ERROR "Provide verified WebP target prefix and generated fixtures")
endif()
add_executable(native_webp main.c)
target_compile_definitions(native_webp PRIVATE WEBP_EXPECTED_VERSION=0x010600)
find_package(WebP CONFIG REQUIRED PATHS "${WEBP_PREFIX}/share/WebP/cmake"
    NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
target_include_directories(native_webp PRIVATE "${WEBP_CASES}")
target_link_libraries(native_webp PRIVATE WebP::webpdemux)
foreach(symbol malloc calloc free realloc)
    target_link_options(native_webp PRIVATE "-Wl,--wrap=${symbol}")
endforeach()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    find_package(Threads REQUIRED)
    target_link_libraries(native_webp PRIVATE Threads::Threads m)
else()
    target_link_options(native_webp PRIVATE "-Wl,--strip-debug")
endif()
