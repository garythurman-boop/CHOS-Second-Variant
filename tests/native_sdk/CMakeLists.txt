cmake_minimum_required(VERSION 3.20)
project(CHOSNativeSDK LANGUAGES C CXX)
include(CheckCSourceCompiles)
include(CheckCXXSourceCompiles)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
check_c_source_compiles("#include <stdlib.h>\nint main(void){void *p=malloc(32);free(p);return 0;}" HAVE_NATIVE_ALLOCATION)
if(NOT HAVE_NATIVE_ALLOCATION)
 message(FATAL_ERROR "real native C allocation link failed")
endif()
check_cxx_source_compiles("#include <thread>\nint main(){std::thread t([]{});t.join();}" HAVE_NATIVE_CXX_THREAD)
if(NOT HAVE_NATIVE_CXX_THREAD)
 message(FATAL_ERROR "real native C++ thread link failed")
endif()
set(CMAKE_REQUIRED_LIBRARIES chos_missing_library_for_negative_check)
check_c_source_compiles("int main(void){return 0;}" WRONG_LIBRARY_ACCEPTED)
if(WRONG_LIBRARY_ACCEPTED)
 message(FATAL_ERROR "try_compile accepted a missing library")
endif()
unset(CMAKE_REQUIRED_LIBRARIES)
check_c_source_compiles("extern int chos_missing_api_for_negative_check(void);int main(void){return chos_missing_api_for_negative_check();}" WRONG_IMPORT_ACCEPTED)
if(WRONG_IMPORT_ACCEPTED)
 message(FATAL_ERROR "try_compile accepted a missing import")
endif()
add_executable(sdk_c main.c)
add_executable(sdk_cpp main.cpp)
