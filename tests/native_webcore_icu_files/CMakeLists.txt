cmake_minimum_required(VERSION 3.20)
project(CHOSBrowserICU LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(component uc i18n)
    add_library(ICU::${component} STATIC IMPORTED)
    set_target_properties(ICU::${component} PROPERTIES IMPORTED_LOCATION "${CHOS_SDK_ROOT}/lib/libicu${component}.a")
endforeach()
add_subdirectory(../../ports/icu/browser-data browser-data)
add_executable(native_browser_icu main.cpp)
target_link_libraries(native_browser_icu PRIVATE chos_browser_icu_data ICU::i18n ICU::uc "${CHOS_SDK_ROOT}/lib/libicustubdata.a")
target_link_options(native_browser_icu PRIVATE "-Wl,--strip-debug")
