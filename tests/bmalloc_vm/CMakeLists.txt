cmake_minimum_required(VERSION 3.20)
project(CHOSBmallocVM LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED WEBKIT_SOURCE)
 message(FATAL_ERROR "Set WEBKIT_SOURCE to the verified pinned checkout")
endif()
add_executable(bmalloc_vm main.cpp)
target_include_directories(bmalloc_vm PRIVATE "${WEBKIT_SOURCE}/Source/bmalloc")
target_compile_definitions(bmalloc_vm PRIVATE NDEBUG=1 USE_SYSTEM_MALLOC=1)
