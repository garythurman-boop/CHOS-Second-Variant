cmake_minimum_required(VERSION 3.20)
project(CHOSNativePNG LANGUAGES C ASM)
if(NOT PNG_PREFIX OR NOT PNG_ARCHIVE OR NOT PNG_CASES)
    message(FATAL_ERROR "Provide the verified codec prefix/archive and generated fixtures")
endif()
add_executable(native_png main.c "${PNG_CASES}/png-fixtures.S")
target_include_directories(native_png PRIVATE "${PNG_PREFIX}/include" "${PNG_CASES}")
target_compile_definitions(native_png PRIVATE Z_SOLO=1)
target_link_libraries(native_png PRIVATE "${PNG_ARCHIVE}" "${PNG_PREFIX}/lib/libz-memory.a")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    find_package(Threads REQUIRED)
    target_link_libraries(native_png PRIVATE Threads::Threads m)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    target_link_options(native_png PRIVATE "-Wl,--strip-debug")
endif()
