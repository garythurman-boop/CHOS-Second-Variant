cmake_minimum_required(VERSION 3.20)
project(CHOSNativeJPEG LANGUAGES C ASM)
if(NOT JPEG_PREFIX OR NOT JPEG_CASES)
    message(FATAL_ERROR "Provide the verified JPEG prefix and generated cases")
endif()
add_executable(native_jpeg main.c "${JPEG_CASES}/jpeg-fixtures.S")
target_include_directories(native_jpeg PRIVATE "${JPEG_PREFIX}/include" "${JPEG_CASES}")
target_link_libraries(native_jpeg PRIVATE "${JPEG_PREFIX}/lib/libjpeg.a")
# Only the reference consumer interposes these platform allocation hooks. The
# original upstream library remains unchanged and still calls its actual backend.
foreach(symbol jpeg_get_small jpeg_get_large jpeg_free_small jpeg_free_large)
    target_link_options(native_jpeg PRIVATE "-Wl,--wrap=${symbol}")
endforeach()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    find_package(Threads REQUIRED)
    target_link_libraries(native_jpeg PRIVATE Threads::Threads m)
else()
    target_link_options(native_jpeg PRIVATE "-Wl,--strip-debug")
endif()
