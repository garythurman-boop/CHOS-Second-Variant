cmake_minimum_required(VERSION 3.20)
project(CHOSPublicSuffix LANGUAGES C)
include("../../ports/publicsuffix/Library.cmake")
file(STRINGS "../../third_party/publicsuffix/tests/tests.txt" cases ENCODING UTF-8)
set(rows "")
foreach(line IN LISTS cases)
    if(line MATCHES "^([^/ ][^ ]*) ([^ ]+)$")
        set(input "${CMAKE_MATCH_1}")
        set(result "${CMAKE_MATCH_2}")
        string(APPEND rows "{\"${input}\",\"${result}\"},\n")
    endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cases.inc" "${rows}")
add_executable(native_publicsuffix main.c)
target_include_directories(native_publicsuffix PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(native_publicsuffix PRIVATE chos_publicsuffix)
