cmake_minimum_required(VERSION 3.20)
project(CHOSJSCCacheGuard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(jsc_cache_guard main.cpp)
target_include_directories(jsc_cache_guard PRIVATE "${ENGINE_BUILD}" "${ENGINE_BUILD}/JavaScriptCore/PrivateHeaders" "${WEBKIT_SOURCE}/Source/JavaScriptCore" "${WEBKIT_SOURCE}/Source/WTF" "${WEBKIT_SOURCE}/Source/bmalloc")
target_compile_definitions(jsc_cache_guard PRIVATE BUILDING_JSCONLY__ BUILDING_WITH_CMAKE=1 HAVE_CONFIG_H=1 BUILDING_WTF=1 NDEBUG=1)
target_link_libraries(jsc_cache_guard PRIVATE "${ENGINE_BUILD}/lib/libWTF.a" "${ENGINE_BUILD}/lib/libbmalloc.a")
