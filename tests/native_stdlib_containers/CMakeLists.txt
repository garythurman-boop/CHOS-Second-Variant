cmake_minimum_required(VERSION 3.20)
project(CHOSNativeStdlibContainers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(native_stdlib_containers main.cpp)
target_compile_options(native_stdlib_containers PRIVATE -fno-builtin)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    find_package(Threads REQUIRED)
    target_link_libraries(native_stdlib_containers PRIVATE Threads::Threads)
endif()
