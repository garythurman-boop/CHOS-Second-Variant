cmake_minimum_required(VERSION 3.20)
project(CHOSNativeCryptoKeys LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT KEY_CASES OR NOT EXISTS "${KEY_CASES}/key-cases.h")
    message(FATAL_ERROR "KEY_CASES must contain independently checked test-only input vectors")
endif()
file(READ "${KEY_CASES}/vectors.json" vectors)
string(JSON expected GET "${vectors}" generated_header sha256)
file(SHA256 "${KEY_CASES}/key-cases.h" actual)
if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Key fixture header differs from its input manifest")
endif()
get_filename_component(CHOS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${CHOS_ROOT}/ports/webcore-crypto-keys/Keys.cmake")
add_executable(native_webcore_crypto_keys main.cpp faults.c)
target_include_directories(native_webcore_crypto_keys PRIVATE "${KEY_CASES}")
target_link_libraries(native_webcore_crypto_keys PRIVATE CHOS::CryptoKeys)
target_link_options(native_webcore_crypto_keys PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
if(KEY_EXPORT_ORACLE)
    if(NOT EXISTS "${KEY_EXPORT_ORACLE}/key-export-oracle.h")
        message(FATAL_ERROR "The export oracle directory is incomplete")
    endif()
    file(READ "${KEY_EXPORT_ORACLE}/export-oracle.json" oracle)
    string(JSON expected GET "${oracle}" vectors_manifest_sha256)
    file(SHA256 "${KEY_CASES}/vectors.json" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Export oracle belongs to another generated key set")
    endif()
    string(JSON expected GET "${oracle}" generated_header_sha256)
    file(SHA256 "${KEY_EXPORT_ORACLE}/key-export-oracle.h" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Export oracle header differs from its manifest")
    endif()
    target_include_directories(native_webcore_crypto_keys PRIVATE "${KEY_EXPORT_ORACLE}")
    target_compile_definitions(native_webcore_crypto_keys PRIVATE KEY_HAVE_EXPORT_ORACLE=1)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    target_link_options(native_webcore_crypto_keys PRIVATE "-Wl,--strip-debug")
else()
    find_package(Threads REQUIRED)
    target_link_libraries(native_webcore_crypto_keys PRIVATE Threads::Threads)
endif()
