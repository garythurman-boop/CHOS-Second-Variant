cmake_minimum_required(VERSION 3.20)
project(CHOSWTFFilesystem LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${ENGINE_BUILD}/lib/libWTF.a")
 message(FATAL_ERROR "Build the actual checked native WTF target first")
endif()
add_executable(wtf_filesystem main.cpp)
target_include_directories(wtf_filesystem PRIVATE "${ENGINE_BUILD}" "${WEBKIT_SOURCE}/Source/WTF" "${WEBKIT_SOURCE}/Source/bmalloc")
target_compile_definitions(wtf_filesystem PRIVATE BUILDING_JSCONLY__ BUILDING_WITH_CMAKE=1 HAVE_CONFIG_H=1 BUILDING_WTF=1 NDEBUG=1)
target_link_libraries(wtf_filesystem PRIVATE "${ENGINE_BUILD}/lib/libWTF.a" "${ENGINE_BUILD}/lib/libbmalloc.a" "${CHOS_SDK_ROOT}/lib/libicuuc.a" "${CHOS_SDK_ROOT}/lib/libicustubdata.a")
