cmake_minimum_required(VERSION 3.20)
project(CHOSWTFStdio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${WEBKIT_SOURCE}/Source/WTF/wtf/FilePrintStream.cpp" OR NOT EXISTS "${ENGINE_BUILD}/cmakeconfig.h")
 message(FATAL_ERROR "Use checked upstream source and real JSCOnly configuration")
endif()
add_executable(wtf_stdio main.cpp "${WEBKIT_SOURCE}/Source/WTF/wtf/FilePrintStream.cpp" "${WEBKIT_SOURCE}/Source/WTF/wtf/PrintStream.cpp" "${WEBKIT_SOURCE}/Source/WTF/wtf/FastMalloc.cpp")
target_include_directories(wtf_stdio PRIVATE "${ENGINE_BUILD}" "${WEBKIT_SOURCE}/Source/WTF" "${WEBKIT_SOURCE}/Source/bmalloc")
target_compile_definitions(wtf_stdio PRIVATE BUILDING_JSCONLY__ BUILDING_WITH_CMAKE=1 HAVE_CONFIG_H=1 BUILDING_WTF=1 NDEBUG=1)
