cmake_minimum_required(VERSION 3.20)
project(CHOSNativeCairo LANGUAGES C ASM)
get_filename_component(CHOS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT GRAPHICS_PREFIX OR NOT FONT_PREFIX)
    message(FATAL_ERROR "Provide verified native or host Cairo and FreeType prefixes")
endif()
add_executable(native_cairo main.c)
target_include_directories(native_cairo PRIVATE "${GRAPHICS_PREFIX}/include/cairo" "${FONT_PREFIX}/include/freetype2")
target_link_libraries(native_cairo PRIVATE "${GRAPHICS_PREFIX}/lib/libcairo.a" "${GRAPHICS_PREFIX}/lib/libpixman-1.a" "${FONT_PREFIX}/lib/libfreetype.a")
if(CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    set(FONT "${CHOS_ROOT}/third_party/text-fixtures/fonts/NotoSans-Regular.ttf")
    set(ASSET "${CMAKE_CURRENT_BINARY_DIR}/font.S")
    file(WRITE "${ASSET}" ".section .rodata.font_asset,\"a\"\n.balign 8\n.global cairo_font_start\n.global cairo_font_end\ncairo_font_start:\n.incbin \"${FONT}\"\ncairo_font_end:\n")
    set_property(SOURCE "${ASSET}" PROPERTY OBJECT_DEPENDS "${FONT}")
    target_sources(native_cairo PRIVATE "${ASSET}")
    target_link_options(native_cairo PRIVATE "-Wl,--strip-debug" "-Wl,-Map,${CMAKE_BINARY_DIR}/native_cairo.map")
else()
    find_package(Threads REQUIRED)
    target_compile_definitions(native_cairo PRIVATE CHOS_CAIRO_HOST=1)
    target_link_libraries(native_cairo PRIVATE Threads::Threads m)
endif()
