cmake_minimum_required(VERSION 3.20)
project(CHOSNativeStdlib LANGUAGES C)
get_filename_component(CHOS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(native_stdlib main.c)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
    find_package(Threads REQUIRED)
    target_compile_definitions(native_stdlib PRIVATE CHOS_STDLIB_HOST=1 rand=chos_test_rand srand=chos_test_srand strdup=chos_test_strdup strndup=chos_test_strndup)
    target_sources(native_stdlib PRIVATE "${CHOS_ROOT}/userland/libc/prng.c" "${CHOS_ROOT}/userland/libc/string_alloc.c")
    target_link_libraries(native_stdlib PRIVATE Threads::Threads)
endif()
