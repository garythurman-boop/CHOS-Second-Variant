cmake_minimum_required(VERSION 3.20)
project(NativeFontconfigFixture LANGUAGES C)
find_package(Fontconfig CONFIG REQUIRED PATHS "${FONTCONFIG_PREFIX}/lib/cmake/Fontconfig" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
add_executable(native_fontconfig main.c)
target_link_libraries(native_fontconfig PRIVATE Fontconfig::Fontconfig)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
 find_package(Threads REQUIRED)
 target_link_libraries(native_fontconfig PRIVATE Threads::Threads m)
else()
 target_link_options(native_fontconfig PRIVATE "-Wl,--strip-debug" "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/native_fontconfig.map")
endif()
