cmake_minimum_required(VERSION 3.20)
project(CHOSNativeSQLite LANGUAGES C)
if(NOT SQLITE_PREFIX)
 message(FATAL_ERROR "Provide verified SQLite target prefix")
endif()
add_executable(native_sqlite main.c)
target_include_directories(native_sqlite PRIVATE "${SQLITE_PREFIX}/include")
target_link_libraries(native_sqlite PRIVATE "${SQLITE_PREFIX}/lib/libsqlite3.a")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
 find_package(Threads REQUIRED)
 target_link_libraries(native_sqlite PRIVATE Threads::Threads m)
endif()
