get_filename_component(_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET CHOS::PALDigest)
    add_subdirectory("${_root}/ports/webcore-crypto" "${CMAKE_CURRENT_BINARY_DIR}/digest")
endif()
set(_receipt "${_root}/docs/dependencies/icu-browser-packages.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_receipt}")
file(READ "${_receipt}" inventory)
string(JSON count LENGTH "${inventory}" files)
math(EXPR last "${count} - 1")
set(entries "")
foreach(i RANGE 0 ${last})
    string(JSON name GET "${inventory}" files ${i} name)
    string(JSON bytes GET "${inventory}" files ${i} bytes)
    string(JSON hash GET "${inventory}" files ${i} sha256)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," octets "${hash}")
    string(APPEND entries "{\"/ICU/${name}\", ${bytes}, {${octets}}},\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/browser-icu-packages.inc" CONTENT "@entries@" @ONLY)
add_library(chos_browser_icu_data STATIC BrowserICUData.c)
target_include_directories(chos_browser_icu_data PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(chos_browser_icu_data PRIVATE CHOS::PALDigest ICU::uc)
