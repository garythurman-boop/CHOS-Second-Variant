cmake_minimum_required(VERSION 3.20)
project(CHOSSQLite VERSION 3.54.1 LANGUAGES C)
if(NOT SQLITE_SOURCE)
 message(FATAL_ERROR "Provide the verified patched amalgamation directory")
endif()
add_library(sqlite3 STATIC "${SQLITE_SOURCE}/sqlite3.c" native_os.c)
target_include_directories(sqlite3 PUBLIC "${SQLITE_SOURCE}")
target_compile_definitions(sqlite3 PRIVATE HAVE_LOCALTIME_R=1 SQLITE_ENABLE_COLUMN_METADATA=1 SQLITE_OS_OTHER=1 SQLITE_MUTEX_CHOS=1 SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_TEMP_STORE=3 SQLITE_ENABLE_API_ARMOR=1 SQLITE_DEFAULT_MEMSTATUS=1 SQLITE_MAX_LENGTH=8388608 SQLITE_MAX_SQL_LENGTH=1048576 SQLITE_MAX_COLUMN=512 SQLITE_MAX_VARIABLE_NUMBER=1024)
set_target_properties(sqlite3 PROPERTIES OUTPUT_NAME sqlite3)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "CHOS")
 find_package(Threads REQUIRED)
 target_link_libraries(sqlite3 PUBLIC Threads::Threads m)
endif()
install(TARGETS sqlite3 ARCHIVE DESTINATION lib)
install(FILES "${SQLITE_SOURCE}/sqlite3.h" "${SQLITE_SOURCE}/sqlite3ext.h" DESTINATION include)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sqlite3.pc" "prefix=${CMAKE_INSTALL_PREFIX}\nlibdir=\${prefix}/lib\nincludedir=\${prefix}/include\nName: SQLite\nDescription: CHOS thread-safe in-memory SQLite core\nVersion: 3.54.1\nLibs: -L\${libdir} -lsqlite3\nCflags: -I\${includedir}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/sqlite3.pc" DESTINATION lib/pkgconfig)
